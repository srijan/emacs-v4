  (use-package tab-bar
    :ensure nil
    :demand t
    ;; :bind
    ;; (("C-<prior>" . tab-bar-switch-to-prev-tab)

    (defun my/tab-bar-find-or-create (name)
      "Switch to tab NAME, creating it if it doesn't exist."
      (unless (seq-some (lambda (tab) (equal (alist-get 'name tab) name))
                        (tab-bar-tabs))
        (tab-bar-new-tab)
        (tab-bar-rename-tab name))
      (tab-bar-switch-to-tab name)
      (when (fboundp 'my/tabspaces-ensure-workspace-scratch)
        (my/tabspaces-ensure-workspace-scratch name)))

    (defun my/mu4e-in-tab (orig-fn &rest args)
      (my/tab-bar-find-or-create "Email")
      (apply orig-fn args))

    (defun my/org-agenda-in-tab (orig-fn &rest args)
      (my/tab-bar-find-or-create "GTD")
      (apply orig-fn args))

    (defun my/elfeed-in-tab (orig-fn &rest args)
      (my/tab-bar-find-or-create "Feeds")
      (apply orig-fn args))

    (advice-add 'mu4e    :around #'my/mu4e-in-tab)
    (advice-add 'notmuch :around #'my/mu4e-in-tab)
    (advice-add 'org-agenda :around #'my/org-agenda-in-tab)
    (advice-add 'elfeed  :around #'my/elfeed-in-tab)

    (defun my/org-file-in-gtd-p (file)
      "Return non-nil when FILE should live in the GTD tab."
      (when (and file (boundp 'org-gtd-directory) org-gtd-directory)
        (file-in-directory-p (expand-file-name file)
                             (file-name-as-directory
                              (expand-file-name org-gtd-directory)))))

    (defun my/tabspaces-project-tab-name (project-root)
      "Return the Tabspaces-style tab name for PROJECT-ROOT."
      (let ((project-root (file-name-as-directory (expand-file-name project-root))))
        (if (and (fboundp 'tabspaces-generate-descriptive-tab-name)
                 (boundp 'tabspaces-project-tab-map)
                 (fboundp 'tabspaces--list-tabspaces))
            (or (cdr (assoc project-root tabspaces-project-tab-map))
                (tabspaces-generate-descriptive-tab-name
                 project-root
                 (tabspaces--list-tabspaces)))
          (file-name-nondirectory
           (directory-file-name project-root)))))

    ;; Route emacsclient opens to the right tab
    (defun my/tab-bar-target-tab (buf)
      "Return a tab name for BUF based on context, or nil."
      (when (buffer-live-p buf)
        (let ((bname (buffer-name buf))
              (mode  (buffer-local-value 'major-mode buf))
              (file  (buffer-file-name buf)))
          (cond
           ((or (string-prefix-p "*mu4e" bname)
                (string-prefix-p "*notmuch" bname)
                (string-prefix-p "*gmi-sync" bname)
                (memq mode '(mu4e-main-mode mu4e-headers-mode
                                            mu4e-view-mode mu4e-compose-mode
                                            notmuch-hello-mode notmuch-search-mode
                                            notmuch-show-mode notmuch-tree-mode
                                            notmuch-message-mode)))
            "Email")
           ((or (string-prefix-p "*elfeed" bname)
                (memq mode '(elfeed-search-mode elfeed-show-mode)))
            "Feeds")
           ((or (equal bname "*Org Agenda*")
                (my/org-file-in-gtd-p file))
            "GTD")
           ((when-let* ((dir (or (and file (file-name-directory file))
                                 (buffer-local-value 'default-directory buf)))
                        (proj (project-current nil dir)))
              (my/tabspaces-project-tab-name (project-root proj))))))))

    (defun my/server-switch-buffer-to-tab (orig-fn &rest args)
      "Route `server-switch-buffer' opens to the appropriate tab."
      (let* ((buf      (car args))
             (tab-name (when buf (my/tab-bar-target-tab buf))))
        (when tab-name
          (my/tab-bar-find-or-create tab-name))
        (apply orig-fn args)))

    (advice-add 'server-switch-buffer :around #'my/server-switch-buffer-to-tab)

    (defun my/protect-main-tab (orig-fn &rest args)
      "Prevent closing the Main tab."
      (if (equal (alist-get 'name (tab-bar--current-tab)) "Main")
          (message "Main tab is protected")
        (apply orig-fn args)))

    (advice-add 'tab-bar-close-tab :around #'my/protect-main-tab)
    (with-eval-after-load 'tabspaces
      (advice-add 'tabspaces-kill-buffers-close-workspace :around #'my/protect-main-tab))

    ;; Route find-file to project tab (create if needed)
    (defun my/find-file-in-project-tab (orig-fn filename &rest args)
      "If FILENAME belongs to a project, switch to (or create) its tab first."
      (let* ((original-tab-name (alist-get 'name (tab-bar--current-tab)))
             (buffer (apply orig-fn filename args))
             (tab-name (my/tab-bar-target-tab buffer)))
        (when tab-name
          (my/tab-bar-find-or-create tab-name)
          (when (and (fboundp 'tabspaces-reset-buffer-list)
                     (not (equal tab-name original-tab-name)))
            (tabspaces-reset-buffer-list))
          (switch-to-buffer buffer))
        buffer))

    (advice-add 'find-file :around #'my/find-file-in-project-tab))

  ;; Tabspaces - workspace management with buffer isolation
  (use-package tabspaces
    :demand t
    :commands (tabspaces-switch-or-create-workspace
               tabspaces-open-or-create-project-and-workspace
               tabspaces-switch-buffer-and-tab)
    :custom
    (tabspaces-use-filtered-buffers-as-default t)
    (tabspaces-default-tab "Main")
    (tabspaces-remove-to-default t)
    (tabspaces-include-buffers '())
    (tabspaces-session nil)
    :config
    (tabspaces-mode 1)

    ;; When killing the last workspace buffer, don't bleed into another workspace.
    ;; switch-to-prev-buffer (called internally on kill-buffer) will skip any
    ;; buffer that isn't local to the current tab.  If no local buffer remains,
    ;; Emacs falls back to *scratch*, which is acceptable.
    (setq switch-to-prev-buffer-skip
          (lambda (_window buffer _bury-or-kill)
            (and (bound-and-true-p tabspaces-mode)
                 (not (tabspaces--local-buffer-p buffer)))))

    (defun my/tabspaces-add-buffer-to-current-tab (buffer)
      "Add BUFFER to the current tab's local buffer list."
      (when (buffer-live-p buffer)
        (set-frame-parameter
         nil 'buffer-list
         (cons buffer (delq buffer (frame-parameter nil 'buffer-list))))))

    (defun my/tabspaces-ensure-workspace-scratch (tab-name)
      "Get or create *scratch-TAB-NAME* and register it in the current tab.
  If the buffer already exists (e.g. from a prior session), it is simply
  linked to the workspace without re-initialising its contents."
      (let* ((buf-name (format "*scratch-%s*" tab-name))
             (buf (get-buffer-create buf-name)))
        (with-current-buffer buf
          (unless (derived-mode-p 'lisp-interaction-mode)
            (lisp-interaction-mode)))
        (my/tabspaces-add-buffer-to-current-tab buf)
        buf))

    (defun my/tabspaces-setup-main-tab ()
      "Ensure the Main tab has shared utility buffers without making them global."
      (tab-bar-rename-tab "Main")
      (my/tabspaces-ensure-workspace-scratch "Main")
      (dolist (buffer-name '("*scratch*" "*Messages*" "*Warnings*"))
        (when-let ((buffer (get-buffer buffer-name)))
          (my/tabspaces-add-buffer-to-current-tab buffer))))

    (my/tabspaces-setup-main-tab)

    ;; Ensure per-workspace scratch exists whenever a workspace is switched/created
    (advice-add 'tabspaces-switch-or-create-workspace :after
                (lambda (&rest _)
                  (my/tabspaces-ensure-workspace-scratch
                   (alist-get 'name (tab-bar--current-tab)))))

    (with-eval-after-load 'consult

        (list :name     "Workspace Buffers"
              :narrow   ?w
              :history  'buffer-name-history
              :category 'buffer
              :state    #'consult--buffer-state
              :default  t
              :items    (lambda () (consult--buffer-query
                                    :predicate #'tabspaces--local-buffer-p
                                    :sort 'visibility
                                    :as #'buffer-name)))
        "Workspace buffer list for consult-buffer.")
      (add-to-list 'consult-buffer-sources 'consult--source-workspace)))

This is my literate Emacs configuration file.

I use minimal-emacs.d as the baseline for my config. It provides two main files:
- early-init.el
- init.el

Apart from these, it allows me to insert configuration as required into the following files:
- pre-early-init.el
- post-early-init.el
- pre-init.el
- post-init.el


When I want to modify my Emacs setup, I edit this file and then
evaluate the following code block or do =C-c C-v C-t=. All files will
be updated accordingly.

  (setq minimal-emacs-ui-features '(context-menu menu-bar dialogs tooltips))

  ;; Reducing clutter in ~/.emacs.d by redirecting files to ~/emacs.d/var/

             (float-time (time-subtract after-init-time before-init-time))
             (time-to-seconds (time-since before-init-time))

  ;;; post-early-init.el --- -*- no-byte-compile: t; lexical-binding: t; -*-
  (when (not (null (getenv "ANDROID_ROOT")))
    (setenv "PATH" (format "%s:%s" "/data/data/com.termux/files/usr/bin"
                           (getenv "PATH")))
    (push "/data/data/com.termux/files/usr/bin" exec-path))

  ;; macOS GUI Emacs does not inherit the shell's PATH, so Homebrew binaries
  ;; (age, gls, ...) are invisible until `exec-path-from-shell' runs -- which
  ;; is far too late: `sj/load-private' needs `age' much earlier in post-init.
  ;; Prepend them here instead; exec-path-from-shell still runs later for the
  ;; rest of the environment.
  (when (eq system-type 'darwin)
    (dolist (dir '("/usr/local/bin" "/opt/homebrew/bin"))
      (when (file-directory-p dir)
        (setenv "PATH" (format "%s:%s" dir (getenv "PATH")))
        (push dir exec-path))))

** Private variables (age)
Personal identifiers (emails, hostnames, employer, sync paths) live encrypted in
=private.json.age= (encrypted with [[https://github.com/FiloSottile/age][age]]) so this file can stay public. The
=defvar= values below are harmless placeholders; =sj/load-private= overrides them
with the real values, decrypted at startup with the age identity at
=~/.config/age/keys.txt=. If =age= or the key is missing, Emacs still loads with
the placeholders instead of breaking.

  (defvar sj/sync-dir "~/Sync"
    "Root of the personal sync folder.")
  (defvar sj/koreader-scp-source "root@192.0.2.10:/mnt/us/koreader/clipboard/")
  (defvar sj/koreader-scp-port "22")

  (defun sj/sync (path)
    "Expand PATH relative to `sj/sync-dir'."
    (expand-file-name path sj/sync-dir))

  (defvar sj/age-identity (expand-file-name "~/.config/age/keys.txt")
    "age identity (private key) used to decrypt `sj/private-file'.")

  (defvar sj/private-file
    ;; Anchor to the repo root (minimal-emacs-user-directory), NOT
    ;; user-emacs-directory: pre-early-init redirects the latter to var/, but
    ;; private.json.age is committed at the repo root.
    (expand-file-name "private.json.age" minimal-emacs-user-directory)
    "age-encrypted JSON holding the real values of the sj/* identifiers.")

  (defun sj/load-private ()
    "Decrypt `sj/private-file' with age and set the sj/* variables it holds.
  A no-op (with a warning) when age or the identity is unavailable, so startup
  never breaks on a machine without the key."
    (cond
     ((not (executable-find "age"))
      (message "sj/load-private: `age' not on PATH; using placeholders"))
     ((not (file-readable-p sj/private-file))
      (message "sj/load-private: %s not readable; using placeholders"
               sj/private-file))
     ((not (file-readable-p sj/age-identity))
      (message "sj/load-private: identity %s not readable; using placeholders"
               sj/age-identity))
     (t
      (condition-case err
          (let ((json (with-temp-buffer
                        (unless (zerop (call-process "age" nil t nil "-d"
                                                     "-i" sj/age-identity
                                                     sj/private-file))
                          (error "age -d failed: %s" (buffer-string)))
                        (goto-char (point-min))
                        (json-parse-buffer :object-type 'alist))))
            (dolist (pair json)
              (set (car pair) (cdr pair))))
        (error (message "sj/load-private: %s" (error-message-string err)))))))

  (sj/load-private)

  (defun android-support-kill-buffer (arg)

    (interactive "p")
    (cond
     ((>= arg 4) (kill-buffer-and-window))
     ((>= arg 1) (kill-buffer))))

  (defun android-support-save-buffer ()
    "Save the current file buffer, or `save-some-buffers' elsewhere."
    (interactive)
    (if (and (buffer-modified-p)
             (or (derived-mode-p 'prog-mode) (derived-mode-p 'text-mode)))

  ;; or reordering a slot is a one-line edit here.
  (defcustom android-support-global-tool-bar-slots
    '((dashboard-open                       . "view-dashboard-outline")
      (org-capture-inbox                    . "inbox-arrow-down-outline")

    ;; Grouped for a single-row layout: arrows, then edit/buffer actions,
    ;; then quit + prefix/chord keys.  The quick slots are appended at setup
    ;; time from `android-support-global-tool-bar-slots'.
    '(;; navigation (hjkl order)

      ("arrow-down" next-line)
      ("arrow-up" previous-line)
      ("arrow-right" forward-char)

      ("circle-multiple-outline" execute-extended-command)
      ("close-circle-multiple-outline" exchange-point-and-mark)
      ("menu" imenu)

  (defun android-support-global-tool-bar-setup ()
    "Rebuild the global `tool-bar-map' from the fixed items and the quick slots."
    (setf tool-bar-map (list 'keymap nil))
    (cl-loop for (icon command key) in android-support-global-tool-bar-items
             do (tool-bar-add-item icon command (or key command)))
    ;; Quick slots: bind the real command directly, the same way the fixed
    ;; items above do (e.g. `consult-buffer').
    (cl-loop for (command . icon) in android-support-global-tool-bar-slots
             do (tool-bar-add-item icon command command)))

    (kill-local-variable 'tool-bar-map))

    "Turn on the Android on-screen toolbar."
    (setq tool-bar-position 'bottom)
    (when (fboundp 'modifier-bar-mode) (modifier-bar-mode 1))
    ;; Make the global toolbar win in every buffer: drop any buffer-local
    ;; `tool-bar-map' that a major mode installs (e.g. the elfeed toolbars).
    ;; `after-change-major-mode-hook' runs after each mode's own hook, so it
    ;; neutralises mode-specific toolbars that would otherwise override ours.
    (add-hook 'after-change-major-mode-hook #'android-support-kill-local-tool-bar-map)
    (android-support-setup-bindings)
    (android-support-global-tool-bar-setup))

             (candidates

                               (pid (format (format "%%%ds" pid-width) (map-elt process 'pid)))

                                    (format "%s %s %s %s" pid user comm (truncate-string-to-width args args-width nil nil t))
                                  (format "%s %s %s" pid user comm))
                                process)))
                      processes))
             (selection (map-elt candidates
                                 (completing-read "kill process: "
                                                  (seq-sort
                                                   (lambda (p1 p2)
                                                     (string-lessp (nth 2 (split-string (string-trim (car p1))))
                                                                   (nth 2 (split-string (string-trim (car p2))))))
                                                   candidates) nil t)))
             (prompt-title (format "%s %s %s"
                                   (map-elt selection 'pid)
                                   (map-elt selection 'user)
                                   (map-elt selection 'comm))))

      (if (member task '("Break" "Reading")) "break" (downcase category)))

    (defun my/clock-timer-dispatch (event task category)
      "Drive `my/clock-timer' for EVENT: one of `in', `out' or `cancel'.
    Uses `call-process' (not a shell string) so titles with quotes/special
    chars are safe, and a destination of 0 so the clock hook never blocks.
    An unhandled backend/event pair yields no URL and is silently skipped."
      (let ((url
             (pcase (cons my/clock-timer event)
               (`(focus . in)
                (format "focus://focus?profile=%s"
                        (url-hexify-string (my/clock-timer-profile task category))))
               (`(focus . ,_)
                (format "focus://unfocus?profile=%s"
                        (url-hexify-string (my/clock-timer-profile task category))))
               (`(session . in)
                (format "session:///start?intent=%s&categoryName=%s"
                        (url-hexify-string task) (url-hexify-string category)))
               (`(session . out) "session:///finish")
               (`(session . cancel) "session:///abandon"))))
        (when (and url (eq system-type 'darwin))
          (call-process "open" nil 0 nil "-g" url))))

    (defvar my/state-file (sj/sync "docs/org/state.json")
      "Synced JSON file publishing Emacs context to an Android KWGT widget.
    Written on org-clock in/out/cancel; read-only on the phone (via Syncthing),
    where Tasker watches this file (File Modified event; a Syncthing write fires
    both Closed-Write and Moved-To, so a single trigger type suffices) and fans
    the keys out to KWGT globals.  Extensible: add keys to the published plist as
    more contextual widgets are built.  Current shape:
    {active, intent, category, started, started_kwgt}.")

    (defun my/state-write (plist)
      "Serialize PLIST to `my/state-file' as one JSON object.
    Errors are swallowed so a sync/IO hiccup never breaks clocking."
      (ignore-errors
        (let ((file (expand-file-name my/state-file)))
          (make-directory (file-name-directory file) t)
          (with-temp-file file
            (insert (json-serialize plist) "\n")))))

    (defun my/clock-area ()
      "Mindwtr area for the entry at point, or nil when it has none.
    The area is rendered as a `CATEGORY' property on the project heading, so
    a task inherits it from an ancestor.  `org-entry-get' can't be used here:
    it special-cases CATEGORY and falls back to the org-data category, i.e.
    the file name (\"mindwtr\") for anything outside a project."
      (org-element-lineage-map (org-element-at-point)
          (lambda (el) (car (org--property-local-values "CATEGORY" nil el)))
        '(headline inlinetask) 'with-self 'first-match))

    (defun my/clock-in ()
      (let ((task-name (org-get-heading t t t t))
            (category (or (my/clock-area) "general")))
        (my/state-write
         (list :active t :intent task-name :category category
               :started (format-time-string "%FT%T%z")
               ;; Kustom-native date text (1955y11M12d22h04m00s) so KWGT's
               ;; $tf()$/$df()$ can read it directly; ISO above is for humans.
               :started_kwgt (format-time-string "%Yy%mM%dd%Hh%Mm%Ss")))
        (setq my/clock-timer--active (cons task-name category))
        (my/clock-timer-dispatch 'in task-name category)))

    (defun my/clock--end (event)
      "Publish an idle state file and tell the focus timer about EVENT."
      (my/state-write '(:active :false))
      (when my/clock-timer--active
        (my/clock-timer-dispatch event
                                 (car my/clock-timer--active)
                                 (cdr my/clock-timer--active))
        (setq my/clock-timer--active nil)))

    (defun my/clock-out () (my/clock--end 'out))
    (defun my/clock-cancel () (my/clock--end 'cancel))

  (defun my/complete-calendar-task ()
    "Copy current calendar task to inbox.org, add logbook entry, and mark done."
    (interactive)
    (let* ((calendar-file (expand-file-name "calendar.org" org-directory))
           (inbox-file (expand-file-name "inbox.org" org-directory))
           (marker (or (org-get-at-bol 'org-marker) (point-marker)))
           (source-file (if marker
                            (buffer-file-name (marker-buffer marker))
                          (buffer-file-name))))
      
      ;; Check if the task is from calendar.org
      (unless (and source-file (string= source-file calendar-file))
        (error "This function should be called on tasks from calendar.org"))
      
      ;; Work with the task at the marker position
      (with-current-buffer (marker-buffer marker)
        (save-excursion
          (goto-char marker)
          (org-back-to-heading t)
          
          ;; Get task information - but limit to current entry only
          (let* ((task-title (org-get-heading t t t t))
                 (entry-start (point))
                 (entry-end (save-excursion (org-end-of-subtree t t)))
                 ;; Extract start and end times from THIS entry only
                 (when-line (save-excursion
                              (goto-char entry-start)
                              (if (re-search-forward "- When: \\(<.*?>\\)--\\(<.*?>\\)" entry-end t)
                                  (list (match-string 1) (match-string 2))
                                nil))))
            
            (unless when-line
              (error "Could not find start and end times in this task"))
            
            (let* ((start-time-angular (car when-line))
                   (end-time-angular (cadr when-line))
                   ;; Convert angular brackets to square brackets for CLOCK entries
                   (start-time-square (replace-regexp-in-string "<\\|>" (lambda (match) (if (string= match "<") "[" "]")) start-time-angular))
                   (end-time-square (replace-regexp-in-string "<\\|>" (lambda (match) (if (string= match "<") "[" "]")) end-time-angular))
                   ;; Calculate duration
                   (start-time (org-parse-time-string start-time-angular))
                   (end-time (org-parse-time-string end-time-angular))
                   (start-seconds (+ (* (nth 2 start-time) 3600) (* (nth 1 start-time) 60)))
                   (end-seconds (+ (* (nth 2 end-time) 3600) (* (nth 1 end-time) 60)))
                   (duration-seconds (- end-seconds start-seconds))
                   (duration-hours (/ duration-seconds 3600))
                   (duration-minutes (/ (% duration-seconds 3600) 60))
                   (duration-string (format "%d:%02d" duration-hours duration-minutes)))
              
              ;; Parse the entry content more carefully
              (goto-char entry-start)
              (forward-line 1) ;; Skip the heading
              (let* ((content-start (point))
                     (properties-start (and (re-search-forward "^:PROPERTIES:$" entry-end t)
                                            (line-beginning-position)))
                     (properties-end (and properties-start
                                          (re-search-forward "^:END:$" entry-end t)
                                          (line-end-position)))
                     (properties-content (and properties-start properties-end
                                              (buffer-substring-no-properties properties-start properties-end)))
                     (remaining-content (and properties-end
                                             (buffer-substring-no-properties (1+ properties-end) entry-end)))
                     ;; Convert angular brackets to square brackets only on "- When:" lines
                     (remaining-content-fixed (and remaining-content
                                                   (replace-regexp-in-string 
                                                    "^- When: <\\([^>]+\\)>--<\\([^>]+\\)>$"
                                                    "- When: [\\1]--[\\2]"
                                                    remaining-content nil nil nil))))
                
                ;; Copy task to inbox.org
                (with-current-buffer (find-file-noselect inbox-file)
                  (goto-char (point-max))
                  (insert "\n* DONE " task-title "\n")
                  ;; Set CLOSED timestamp to match event end time
                  (insert "CLOSED: " end-time-square "\n")
                  
                  ;; Insert properties drawer if it exists
                  (when properties-content
                    (insert properties-content "\n"))
                  
                  ;; Insert logbook
                  (insert ":LOGBOOK:\n")
                  (insert "CLOCK: " start-time-square "--" end-time-square " => " duration-string "\n")
                  (insert ":END:\n")
                  
                  ;; Insert remaining content (the body after properties) with fixed When line
                  (when remaining-content-fixed
                    (insert remaining-content-fixed))

                  (my/org-calendar-html-to-text)
                  
                  ;; Save the inbox file
                  (save-buffer)
                  
                  ;; Position cursor at the new task
                  (org-back-to-heading t)))
              
              (message "Calendar task copied to inbox.org and marked as DONE with logbook entry")))))))

  (defun sj-linkding--fetch-tags ()
    "Fetch all tags from Linkding, update `sj-linkding--tags-cache'."
    (unless (and sj-linkding-url sj-linkding-token)
      (user-error "Set sj-linkding-url and sj-linkding-token first"))
    (let* ((endpoint (concat (string-trim-right sj-linkding-url "/")

      (when buf

          (goto-char (point-min))

                 (data (json-read)))

                  (mapcar (lambda (tag) (alist-get 'name tag))
                          (alist-get 'results data)))))
        (kill-buffer buf))))

    "Return cached Linkding tags, fetching from server if cache is empty or REFRESH."
    (when (or refresh (null sj-linkding--tags-cache))

    sj-linkding--tags-cache)

#+begin_src emacs-lisp :tangle "post-init.el"
  (use-package clatter
    :commands clatter-connect
    :bind (:map clatter-mode-map
                ("C-c ."   . clatter-track-switch)
                ("C-c C-." . clatter-track-switch)
                ("C-c :"   . clatter-track-list)
                ("C-c n"   . clatter-nicklist-toggle)
                ("C-c C-n" . clatter-nicklist-toggle)
                ("C-c b"   . my/clatter-status)
                ("C-c C-p" . clatter-chathistory-more))
    :config
    (clatter-setup)
    (require 'gnutls)

    ;; Root-cause robustness: clatter's process filter dispatches each line
    ;; with no per-line error isolation, so a single signalling handler
    ;; aborts the `while' loop and drops every remaining buffered line.  soju
    ;; replays a network's channel JOINs as one burst, so one bad line early
    ;; in the burst (e.g. an empty-body message) truncates it and most
    ;; channel buffers never get created.  Isolate each line: log and carry
    ;; on so the rest of the burst still lands.
    (define-advice clatter--handle-line
        (:around (orig conn line) sj/isolate-line-errors)
      (condition-case err
          (funcall orig conn line)
        (error
         (message "[clatter] handler error on %S: %S" line err))))

    ;; soju's CHATHISTORY compares timestamp bounds at whole-second granularity
    ;; (and rejects msgid bounds), so paging re-includes the boundary second and
    ;; the renderer would draw the boundary message twice.  Dedup the RENDER by
    ;; msgid via an in-memory per-channel set: the redrawn boundary is dropped
    ;; from display, while the logs stay complete (logging dedups separately via
    ;; the high-water mark).  Also contains render errors so a failure can't
    ;; abort the other batch hooks (logging, paging).
    (defvar my/clatter--last-rendered-msgid (make-hash-table :test 'equal)
      "Map (network . downcased-target) -> msgid of the newest rendered message.
The boundary for dropping already-seen messages: CHATHISTORY returns messages
in chronological order, and soju re-returns the boundary second on the next
page, so everything up to and including this msgid was already drawn.")

    (defun my/clatter--after-msgid (messages boundary)
      "Return the tail of MESSAGES (chronological) after the one whose msgid is
BOUNDARY -- everything up to and including BOUNDARY was already rendered.  If
BOUNDARY is nil or not present (no overlap), return MESSAGES unchanged."
      (if (null boundary)
          messages
        (let ((tail messages) (found nil))
          (while (and tail (not found))
            (when (equal (my/clatter--msg-msgid (car tail)) boundary)
              (setq found t))
            (setq tail (cdr tail)))
          (if found tail messages))))

    (define-advice clatter-ui--on-batch-complete
        (:around (orig conn batch-type target messages) sj/render-dedup)
      (condition-case _
          (if (and (member batch-type '("chathistory" "draft/chathistory"))
                   (stringp target))
              (let* ((key (cons (clatter-connection-network-id conn)
                                (downcase target)))
                     (fresh (my/clatter--after-msgid
                             messages
                             (gethash key my/clatter--last-rendered-msgid))))
                (when fresh
                  (let ((last-id (my/clatter--msg-msgid (car (last fresh)))))
                    (when last-id
                      (puthash key last-id my/clatter--last-rendered-msgid))))
                (funcall orig conn batch-type target fresh))
            (funcall orig conn batch-type target messages))
        (error nil)))

    ;; Root cause of the `(args-out-of-range N N)' errors during history
    ;; playback: `clatter--find-message-by-msgid' reads the reply target's
    ;; sender/text/type with `get-text-property FOUND ...' but WITHOUT
    ;; re-entering the channel buffer FOUND came from.  During playback the
    ;; current buffer is not the channel buffer, so a reply message whose
    ;; target position exceeds the current buffer's `point-max' throws (and
    ;; even when it doesn't, the preview is read from the wrong buffer).  Read
    ;; in the buffer the position belongs to.
    (define-advice clatter--find-message-by-msgid
        (:override (buffer msgid) sj/read-reply-in-right-buffer)
      (when-let* ((found (clatter--find-message-position-by-msgid buffer msgid)))
        (with-current-buffer buffer
          (cons (cons (get-text-property found 'clatter-sender)
                      (get-text-property found 'clatter-text))
                (get-text-property found 'clatter-msg-type)))))

    ;; Upstream crashes on a NOTICE/PRIVMSG with an empty body: `raw-text' is
    ;; nil, so `clatter-insert-generic' hits (get-text-property 0 ... nil) and
    ;; the process filter aborts (seen on multi-network connect, where empty
    ;; server NOTICEs interrupt CHATHISTORY playback).  Coerce a nil
    ;; sender/text to safe defaults.  No-ops once upstream guards this.
    (define-advice clatter-insert-generic
        (:filter-args (args) sj/nil-text-guard)
      (let ((args (copy-sequence args)))
        (unless (nth 2 args) (setf (nth 2 args) "*"))
        (unless (nth 3 args) (setf (nth 3 args) ""))
        args))

    ;; On (re)connect fetch the whole gap since we were last here, not just the
    ;; last `clatter-chathistory-limit' messages.  Upstream only gap-fills when
    ;; an in-memory cursor exists (lost on restart) and otherwise fetches
    ;; LATEST N.  We fetch AFTER the more recent of the last-seen cursor and a
    ;; floor `my/clatter-chathistory-max-age-days' days back -- but a single
    ;; CHATHISTORY AFTER is capped (soju returns ~100 per request, oldest
    ;; first), so a busy channel would only replay the oldest 100 in the
    ;; window.  Page forward from the newest message of each batch until a
    ;; channel is caught up (empty batch / no forward progress) or a safety cap.
    (defvar my/clatter-chathistory-max-age-days 1
      "Hard floor, in days, on how far back any on-join fetch may reach.
This is the cap for a mid-session reconnect gap-fill; it is NOT the fresh-start
display window (see `my/clatter-startup-history-hours').")
    (defvar my/clatter-startup-history-hours 4
      "How much history to replay from soju on a FRESH clatter start.
On first join of a channel this session the buffer is prefilled with this
many hours from the bouncer, regardless of what was already seen/logged before
the last stop -- so opening clatter at 3pm shows history back to 11am, not just
the gap since you quit.  Capped by `my/clatter-chathistory-max-age-days'.")
    (defvar my/clatter-chathistory-max-pages 10
      "Safety cap on CHATHISTORY pages fetched per channel during gap catch-up.
At ~100 messages per page this bounds a single channel's replay to ~10k,
enough for a full day on all but pathologically busy channels.  Rendering
is synchronous, so a very large fill can be briefly laggy while it lands.")
    (defvar my/clatter--chathistory-catchup (make-hash-table :test 'equal)
      "In-progress gap catch-up: (network . target) -> (PAGE . LAST-SINCE-TIME).")

    (defun my/clatter--gap-key (conn target)
      (cons (clatter-connection-network-id conn) (downcase target)))

    (defun my/clatter--newest-msg (messages)
      "Return the MESSAGE plist with the newest :time in MESSAGES, or nil."
      (let (best best-time)
        (dolist (m messages best)
          (let ((tm (plist-get m :time)))
            (when (and tm (or (null best-time) (time-less-p best-time tm)))
              (setq best m best-time tm))))))

    (defun my/clatter--msg-msgid (msg)
      "Return MSG's msgid (from the `clatter-msgid' text property on :text), or nil."
      (let ((text (plist-get msg :text)))
        (and (stringp text) (> (length text) 0)
             (get-text-property 0 'clatter-msgid text))))

    (defun my/clatter--chathistory-req-limit (conn)
      "CHATHISTORY page size to request on CONN: `clatter-chathistory-limit'
clamped to the server's advertised CHATHISTORY max (ISUPPORT).  Using the
real cap makes a short page reliably mean \"caught up\" rather than \"server
capped a full page below what we asked\" -- so pagination neither under-fills
nor fetches an extra page that just re-renders the boundary message."
      (let* ((isupport (clatter-connection-isupport conn))
             (max (and isupport (gethash "CHATHISTORY" isupport)))
             (max (and (stringp max) (string-to-number max))))
        (if (and max (> max 0))
            (min clatter-chathistory-limit max)
          clatter-chathistory-limit)))

    (defun my/clatter--gap-fetch (conn target selector page last-time)
      "Request the PAGE'th gap page for TARGET via `CHATHISTORY AFTER SELECTOR'.
SELECTOR is a preformatted CHATHISTORY bound (\"msgid=...\" from the previous
page's newest message, or \"timestamp=...\" for the first page).  LAST-TIME is
the newest message time seen so far, stashed for the progress guard.  Msgid
bounds are excluded precisely by soju, so paging never re-renders the boundary
message (unlike a timestamp bound, which soju compares at whole-second
granularity)."
      (when (clatter-chathistory--available-p conn)
        (puthash (my/clatter--gap-key conn target) (cons page last-time)
                 my/clatter--chathistory-catchup)
        (clatter-send
         conn
         (format "CHATHISTORY AFTER %s %s %d"
                 target selector (my/clatter--chathistory-req-limit conn)))))

    (defun my/clatter--gap-timestamp-selector (time)
      "Return a CHATHISTORY `timestamp=' selector for TIME (millisecond precision)."
      (format "timestamp=%s"
              (format-time-string "%Y-%m-%dT%H:%M:%S.%3NZ" time t)))

    (defun my/clatter--gap-continue (conn batch-type target messages)
      "Page CHATHISTORY forward until a catching-up channel is current.
Continues only for channels started by the on-join gap fetch, and only after
a FULL page (count == the server's real per-request limit); a short page means
we have everything, so we stop.  Pages by the newest message's MSGID, which
soju excludes precisely -- so no boundary message is re-rendered and no
same-second message is dropped.  Falls back to a millisecond timestamp bound
if the newest message carries no msgid.  The `newest' time progress check is a
secondary guard against a non-advancing cursor.  Wrapped in `condition-case'
so a paging error can't abort the other batch hooks (they run before the
renderer)."
      (condition-case err
          (when (and (member batch-type '("chathistory" "draft/chathistory"))
                     (stringp target))
            (when-let* ((key (my/clatter--gap-key conn target))
                        (state (gethash key my/clatter--chathistory-catchup)))
              (let* ((page (car state))
                     (last-time (cdr state))
                     (newest-msg (my/clatter--newest-msg messages))
                     (newest (and newest-msg (plist-get newest-msg :time)))
                     (count (length messages)))
                (if (and (>= count (my/clatter--chathistory-req-limit conn))
                         newest (time-less-p last-time newest)
                         (< page my/clatter-chathistory-max-pages))
                    ;; Page by TIMESTAMP -- soju rejects msgid bounds.  It
                    ;; truncates to whole seconds and re-includes the boundary
                    ;; second; the render dedup drops the redrawn boundary.
                    (my/clatter--gap-fetch
                     conn target (my/clatter--gap-timestamp-selector newest)
                     (1+ page) newest)
                  (remhash key my/clatter--chathistory-catchup)
                  (when (>= page my/clatter-chathistory-max-pages)
                    (message "[clatter] %s: history catch-up hit %d-page cap"
                             target my/clatter-chathistory-max-pages))))))
        (error (message "[clatter] gap-continue error on %s: %S" target err))))
    (add-hook 'clatter-batch-complete-hook #'my/clatter--gap-continue)

    (define-advice clatter-chathistory--on-join
        (:override (conn sender channel _account _realname) sj/gap-since-last-connect)
      (when (and clatter-chathistory-enabled
                 clatter-chathistory-on-join
                 (string-equal (clatter-prefix-nick sender)
                               (clatter-connection-nick conn))
                 (clatter-chathistory--available-p conn))
        (let* ((network (clatter-connection-network-id conn))
               (buf (clatter-get-buffer network channel))
               ;; In-memory cursor: non-nil only when this buffer already holds
               ;; messages THIS session -- i.e. a mid-session reconnect.  On a
               ;; fresh clatter start the buffer is new, so this is nil.
               (mem-last (and buf (buffer-local-value
                                   'clatter-chathistory--last-timestamp buf)))
               (mem-secs (and mem-last (float-time mem-last)))
               ;; Hard floor: never fetch further back than this.
               (floor-secs (float-time
                            (time-subtract (current-time)
                                           (days-to-time my/clatter-chathistory-max-age-days))))
               (since-secs
                (if mem-secs
                    ;; Reconnect within this session: fill only the gap since the
                    ;; newest message already in the buffer (floored).  Live
                    ;; render-dedup drops any boundary message soju re-includes.
                    (max mem-secs floor-secs)
                  ;; Fresh start: replay the whole startup window from soju,
                  ;; which retains the backlog.  The persisted log high-water is
                  ;; deliberately NOT consulted here -- it gates on-disk log
                  ;; dedup, not the display -- so history from before the last
                  ;; stop is shown again (3pm-yesterday .. now, not just the gap).
                  (max floor-secs
                       (float-time
                        (time-subtract (current-time)
                                       (seconds-to-time
                                        (* my/clatter-startup-history-hours 3600)))))))
               (since (seconds-to-time since-secs)))
          ;; Fresh start wants the FULL window: clear any stale render-dedup
          ;; boundary (e.g. left by a same-session `my/clatter-stop') so
          ;; `my/clatter--after-msgid' does not drop the older half of it.
          (unless mem-secs
            (remhash (cons network (downcase channel))
                     my/clatter--last-rendered-msgid))
          (my/clatter--gap-fetch conn channel
                                 (my/clatter--gap-timestamp-selector since)
                                 0 since))))

    ;; Batch playback (`clatter-ui--on-batch-complete') inserts history by
    ;; calling `clatter-insert-privmsg' directly, bypassing `clatter-privmsg-hook'
    ;; where the log writer lives -- so replayed CHATHISTORY never reaches the
    ;; on-disk logs.  Log it here, gated by a persistent per-target high-water
    ;; mark (the newest message time already logged) so nothing is written
    ;; twice, even across restarts.  Live logging advances the mark; history
    ;; logging is gated by it and only writes strictly-newer messages.
    (defvar my/clatter-log-highwater-file
      (expand-file-name "clatter/log-highwater.el" user-emacs-directory)
      "File persisting the newest-logged message time per (network . target).")
    (defvar my/clatter--log-hw (make-hash-table :test 'equal)
      "Map (network . downcased-target) -> (SECS . MSGID) of the newest logged
message: SECS (float seconds) gates logging dedup, MSGID enables precise
restart resume.")
    (defvar my/clatter--log-hw-dirty nil
      "Non-nil when `my/clatter--log-hw' has unsaved changes.")

    (defun my/clatter--log-hw-load ()
      "Load the persisted log high-water marks from disk."
      (when (file-readable-p my/clatter-log-highwater-file)
        (with-temp-buffer
          (insert-file-contents my/clatter-log-highwater-file)
          (ignore-errors
            (dolist (cell (read (current-buffer)))
              ;; Value is (SECS . MSGID); migrate an old bare-float SECS entry.
              (puthash (car cell)
                       (if (consp (cdr cell)) (cdr cell) (cons (cdr cell) nil))
                       my/clatter--log-hw))))))

    (defun my/clatter--log-hw-save ()
      "Persist the log high-water marks to disk if changed."
      (when my/clatter--log-hw-dirty
        (let (alist)
          (maphash (lambda (k v) (push (cons k v) alist)) my/clatter--log-hw)
          (with-temp-file my/clatter-log-highwater-file
            (let ((print-length nil) (print-level nil))
              (prin1 alist (current-buffer))))
          (setq my/clatter--log-hw-dirty nil))))

    (defun my/clatter--log-hw-bump (network target time &optional msgid)
      "Advance NETWORK/TARGET's high-water mark to (TIME . MSGID) when newer."
      (when (and (stringp target) time)
        (let* ((key (cons network (downcase target)))
               (secs (float-time time))
               (cur-secs (car (gethash key my/clatter--log-hw))))
          (when (or (null cur-secs) (> secs cur-secs))
            (puthash key (cons secs msgid) my/clatter--log-hw)
            (setq my/clatter--log-hw-dirty t)))))

    (defun my/clatter--log-history (conn batch-type target messages)
      "Write CHATHISTORY playback MESSAGES for TARGET to the local logs.
Only messages newer than the persisted high-water mark are written.
Wrapped in `condition-case' so a log-write error can't abort the other
batch hooks (this one runs before gap paging and the renderer)."
      (condition-case err
          (when (and (member batch-type '("chathistory" "draft/chathistory"))
                     (stringp target))
            (let* ((network (clatter-connection-network-id conn))
                   (hw-secs (car (gethash (cons network (downcase target))
                                          my/clatter--log-hw))))
              (dolist (msg messages)
                (let* ((sender (plist-get msg :sender))
                       (text (plist-get msg :text))
                       (time (plist-get msg :time))
                       (secs (and time (float-time time))))
                  ;; Compare every message to the pre-batch mark: all
                  ;; genuinely-new messages log; `-bump' advances the mark.
                  (when (and (stringp sender) (stringp text)
                             (or (null secs) (null hw-secs) (> secs hw-secs)))
                    ;; `clatter-log--write' honors `clatter-log-enable' and
                    ;; `clatter-log-exclude-targets' itself.
                    (clatter-log--write
                     network target
                     (if (eq (plist-get msg :type) 'action)
                         (format "* %s %s" sender text)
                       (format "<%s> %s" sender text))
                     time)
                    (my/clatter--log-hw-bump network target time
                                             (my/clatter--msg-msgid msg)))))))
        (error (message "[clatter] log-history error on %s: %S" target err))))

    (defun my/clatter--log-hw-track-live (conn sender target text server-time)
      "Advance the high-water mark for a live message just logged.
Keeps history replay from re-logging what we already saw in real time, and
records the message's msgid so a restart can resume precisely by msgid."
      (let* ((network (clatter-connection-network-id conn))
             (my-nick (clatter-connection-nick conn))
             (log-target (if (clatter-channel-name-p target)
                             target
                           (if (string-equal target my-nick)
                               (clatter-prefix-nick sender)
                             target)))
             (msgid (and (stringp text) (> (length text) 0)
                         (get-text-property 0 'clatter-msgid text))))
        (my/clatter--log-hw-bump network log-target
                                 (or server-time (current-time)) msgid)))

    (my/clatter--log-hw-load)
    (add-hook 'clatter-batch-complete-hook #'my/clatter--log-history)
    (advice-add 'clatter-log--on-privmsg :after #'my/clatter--log-hw-track-live)
    (advice-add 'clatter-log--on-action  :after #'my/clatter--log-hw-track-live)
    (advice-add 'clatter-log-flush :after #'my/clatter--log-hw-save)
    (add-hook 'kill-emacs-hook #'my/clatter--log-hw-save)

    ;; `clatter-get-or-create-buffer' returns whatever the registry
    ;; (`clatter--buffer-alist') holds without checking `buffer-live-p', so a
    ;; stale entry pointing at a killed buffer makes every handler fail with
    ;; "Selecting deleted buffer".  Treat a dead buffer as absent and prune it,
    ;; so the registry self-heals and the buffer is recreated fresh.
    (define-advice clatter-get-buffer
        (:around (orig network target) sj/skip-dead-buffers)
      (let ((buf (funcall orig network target)))
        (if (and buf (not (buffer-live-p buf)))
            (progn (clatter-remove-buffer network target) nil)
          buf)))

    ;; Cleanly stop clatter without detaching any channels in soju.  Killing a
    ;; channel buffer normally sends PART (see `clatter-cmd-close'), which soju
    ;; turns into a detach.  `clatter-disconnect-all' sends QUIT and clears each
    ;; connection's process, so the kept PART-on-kill hook sends no PART (it is
    ;; guarded by a live process) yet still runs `clatter-remove-buffer' to
    ;; prune the registry.  Use before upgrading clatter: never upgrade live.
    (defun my/clatter-stop ()
      "Disconnect every clatter network and kill all clatter buffers.
No channels are detached: connections QUIT (not PART) and the buffer
registry is pruned so no dead buffers are left behind."
      (interactive)
      (clatter-disconnect-all)
      (dolist (b (buffer-list))
        (when (buffer-live-p b)
          (with-current-buffer b
            (when (derived-mode-p 'clatter-mode)
              (kill-buffer b)))))
      (setq clatter--buffer-alist nil)
      (message "clatter: stopped, all buffers killed (no channels detached)"))

    ;; Show an ERC/weechat-style "unread from here" line when jumping to a
    ;; channel.  Upstream only draws the read-marker overlay on an INCOMING
    ;; server MARKREAD and clears it on focus (`clatter-read-marker-auto-send'),
    ;; so a normal jump shows nothing.  The jump commands (clatter-track-switch,
    ;; the consult source, the activity list) all call `clatter-clear-activity'
    ;; with the unread count still set, so draw the boundary line there -- just
    ;; before it resets -- then let it persist (auto-send is nil below).
    ;; `clatter-read-marker--update-visual' positions purely from
    ;; `clatter--unread-count' and only gates on a non-nil `--msgid', so seed one.
    (defun my/clatter--unread-boundary-pos ()
      "Return the buffer position where the unread-marker line belongs, or nil.
Persisted `clatter--last-read-time' (restored from `clatter-read-state-file'
on restart) marks the boundary; the overlay's `before-string' is drawn just
ABOVE the returned position.  Order-aware:
- oldest-first (unread at the bottom): the first message NEWER than last-read.
- newest-first (unread at the top): the first message NOT newer than last-read
  (i.e. the first already-read one), so the line lands below the unread block.
Scans by the `clatter-server-time' text property -- exact and multi-line safe.
Returns nil if there is no persisted read time to anchor on."
      (when clatter--last-read-time
        (let ((oldest-first (eq clatter-message-order 'oldest-first)))
          (save-excursion
            (goto-char (point-min))
            (let ((pos nil))
              (while (and (not pos) (< (point) (point-max)))
                (let ((st (get-text-property (point) 'clatter-server-time)))
                  (cond
                   ((null st)
                    (goto-char (or (next-single-property-change
                                    (point) 'clatter-server-time nil (point-max))
                                   (point-max))))
                   ((if oldest-first
                        (time-less-p clatter--last-read-time st)
                      (not (time-less-p clatter--last-read-time st)))
                    (setq pos (line-beginning-position)))
                   (t
                    (goto-char (or (next-single-property-change
                                    (point) 'clatter-server-time nil (point-max))
                                   (point-max)))))))
              pos)))))

    (defun my/clatter--place-unread-overlay (buffer)
      "Draw the \"unread from here\" line in BUFFER, or clear it when caught up.
Positions by persisted read time when possible, else falls back to counting
`clatter--unread-count' messages from the appropriate end (order-aware)."
      (when (buffer-live-p buffer)
        (with-current-buffer buffer
          (when (and (derived-mode-p 'clatter-mode) clatter-read-marker-enabled)
            (when clatter-read-marker--overlay
              (delete-overlay clatter-read-marker--overlay)
              (setq clatter-read-marker--overlay nil))
            (let ((pos (when (> (or clatter--unread-count 0) 0)
                         (or (my/clatter--unread-boundary-pos)
                             (save-excursion
                               (if (eq clatter-message-order 'oldest-first)
                                   (progn
                                     (goto-char (point-max))
                                     (forward-line (- clatter--unread-count)))
                                 (goto-char (or clatter--messages-marker (point-min)))
                                 (forward-line clatter--unread-count))
                               (line-beginning-position))))))
              (when pos
                (let ((ov (make-overlay pos (1+ pos))))
                  (overlay-put ov 'before-string
                               (propertize (concat (make-string 40 ?\x2500) "\n")
                                           'face 'clatter-read-marker-line))
                  (overlay-put ov 'clatter-read-marker t)
                  (setq clatter-read-marker--overlay ov))))))))

    ;; Draw on jump: the jump commands (clatter-track-switch, consult source,
    ;; activity list, status buffer) all call `clatter-clear-activity' with the
    ;; unread count still set, so draw the boundary just before it resets, then
    ;; let it persist (auto-send is nil below, so focus does not wipe it).
    (defun my/clatter--draw-unread-marker (buffer &rest _)
      "Advice for `clatter-clear-activity': draw the unread boundary line."
      (my/clatter--place-unread-overlay buffer))
    (advice-add 'clatter-clear-activity :before #'my/clatter--draw-unread-marker)

    ;; Draw after history replay too, so on restart the marker is already there
    ;; when you open a channel -- without needing a jump.  Fires per history
    ;; page; the boundary is stable once the first post-read message arrives.
    (defun my/clatter--marker-after-history (conn batch-type target _messages)
      "Advice/hook: place the unread marker once history for TARGET replays.
Wrapped in `condition-case': `add-hook' prepends, so this runs before
`my/clatter--gap-continue' / `my/clatter--log-history' on the batch hook,
and an unguarded error here would abort history paging and logging."
      (condition-case err
          (when (and (member batch-type '("chathistory" "draft/chathistory"))
                     (stringp target))
            (let ((buf (clatter-get-buffer
                        (clatter-connection-network-id conn) target)))
              (when (and buf (not (eq buf (current-buffer))))
                (my/clatter--place-unread-overlay buf))))
        (error (message "[clatter] marker-after-history error on %s: %S"
                        target err))))
    (add-hook 'clatter-batch-complete-hook #'my/clatter--marker-after-history)

    ;; --- weechat/irssi-style status buffer ---------------------------------
    ;; A persistent roster of EVERY network and channel/query with unread
    ;; indicators.  Upstream `clatter-track-list' (*clatter-activity*) only
    ;; lists targets that currently have activity, so it is empty when things
    ;; are quiet; this always shows the full tree.  Grouped by network (server
    ;; buffer first, then channels, then queries); RET / mouse-1 jumps and
    ;; clears.  Redraws off clatter's existing track timer (advising
    ;; `clatter-track--update'), so there is no extra timer and it only rebuilds
    ;; when the buffer is actually displayed.
    (defvar my/clatter-status-buffer-name "*clatter-status*"
      "Name of the persistent clatter status/roster buffer.")

    (defun my/clatter-status--sort-key (buf)
      "Sort key (TYPE-RANK . NAME) for BUF within its network."
      (with-current-buffer buf
        (cons (pcase clatter--buffer-type ('server 0) ('channel 1) (_ 2))
              (downcase (or clatter--target "")))))

    (defun my/clatter-status--line-less (a b)
      "Order clatter buffers A and B: server, channels, queries; then name."
      (let ((ka (my/clatter-status--sort-key a))
            (kb (my/clatter-status--sort-key b)))
        (if (= (car ka) (car kb)) (string< (cdr ka) (cdr kb))
          (< (car ka) (car kb)))))

    (defun my/clatter-status--line (buf)
      "Return a propertized roster line string for clatter BUF.
Reads BUF's locals but does NOT insert -- the caller inserts it into the
status buffer, so this must never run `insert' inside BUF."
      (with-current-buffer buf
        (let* ((unread (or clatter--unread-count 0))
               (mention clatter--has-mention)
               (server (eq clatter--buffer-type 'server))
               (name (or clatter--target "*server*"))
               (type (cond (mention 'mention)
                           ((eq clatter--buffer-type 'query) 'dm)
                           (t 'activity)))
               (marker (cond (mention "@") ((> unread 0) "*") (t " ")))
               (count (if (> unread 0) (format " (%d)" unread) ""))
               (face (cond ((> unread 0) (clatter-track--face type nil))
                           (server 'shadow)
                           (t 'default))))
          ;; Keep the trailing newline UNpropertized: `mouse-face' highlights
          ;; the maximal run sharing its value, so a propertized newline would
          ;; fuse a whole network group into one hover region.
          (concat (propertize (format "  %s %s%s" marker name count)
                              'face face
                              'my/clatter-status-buffer buf
                              'mouse-face 'highlight
                              'help-echo (format "%d unread" unread))
                  "\n"))))

    (defun my/clatter-status--active-channel ()
      "The clatter buffer currently shown in the main content area, or nil."
      (let ((w (my/clatter--content-window)))
        (when (window-live-p w)
          (let ((b (window-buffer w)))
            (when (and (buffer-live-p b)
                       (buffer-local-value 'clatter--target b))
              b)))))

    (defun my/clatter-status--goto-buffer (target)
      "Move point to the roster line for clatter buffer TARGET.
Return non-nil if found (leaving point on that line); nil otherwise."
      (goto-char (point-min))
      (let (found)
        (while (and (not found) (not (eobp)))
          (if (eq (get-text-property (point) 'my/clatter-status-buffer) target)
              (setq found t)
            (forward-line 1)))
        found))

    (defvar-local my/clatter-status--hl-overlay nil
      "Overlay highlighting the active channel's roster line, treemacs-style.")

    (defun my/clatter-status--highlight (buf)
      "Highlight the roster line for the active channel BUF (or clear it)."
      (when my/clatter-status--hl-overlay
        (delete-overlay my/clatter-status--hl-overlay)
        (setq my/clatter-status--hl-overlay nil))
      (when (buffer-live-p buf)
        (save-excursion
          (when (my/clatter-status--goto-buffer buf)
            (let ((ov (make-overlay (line-beginning-position)
                                    (min (point-max) (1+ (line-end-position))))))
              (overlay-put ov 'face 'hl-line)
              (setq my/clatter-status--hl-overlay ov))))))

    (defun my/clatter-status-refresh (&rest _)
      "Redraw the clatter status buffer if it is live.
Keeps the cursor on the same channel across redraws (by buffer identity, not
raw position) and keeps a treemacs-style highlight on the active channel.
When the sidebar is not the selected window the cursor follows the active
channel; while you are browsing it (sidebar selected) your point is kept.
Runs off the `clatter-track--update' advice (the message/activity path), so
guard it: an unguarded error would propagate into whoever bumped activity."
      (interactive)
      (condition-case err
          (let ((buf (get-buffer my/clatter-status-buffer-name)))
            (when (buffer-live-p buf)
              (with-current-buffer buf
                (let* ((inhibit-read-only t)
                       (win (get-buffer-window buf))
                       (sidebar-selected (and win (eq (selected-window) win)))
                       (active (my/clatter-status--active-channel))
                       (pos (point))
                       (point-buf (if sidebar-selected
                                      (get-text-property
                                       (point) 'my/clatter-status-buffer)
                                    active))
                       (groups (make-hash-table :test 'equal))
                       (order nil))
                  (erase-buffer)
                  (dolist (b (clatter-all-buffers))
                    (let ((net (buffer-local-value 'clatter--network b)))
                      (unless (member net order) (push net order))
                      (push b (gethash net groups))))
                  (dolist (net (sort order #'string<))
                    (insert (propertize (format "%s\n" net) 'face '(bold underline)))
                    (dolist (b (sort (gethash net groups)
                                     #'my/clatter-status--line-less))
                      (insert (my/clatter-status--line b)))
                    (insert "\n"))
                  (unless (and point-buf (my/clatter-status--goto-buffer point-buf))
                    (goto-char (min pos (point-max))))
                  (my/clatter-status--highlight active)))))
        (error (message "[clatter] status-refresh error: %S" err))))

    (defcustom my/clatter-status-width 24

      :type 'integer :group 'clatter)
    (defcustom my/clatter-channel-open-style 'reuse
      "How opening a channel from the sidebar uses the main (non-side) area.
`reuse' shows it in the single main content window (full remaining space,
replacing the previous channel); `split' splits the main window so the new
channel opens alongside the current one."
      :type '(choice (const :tag "Reuse the main window (full width)" reuse)

    (defun my/clatter--content-window ()
      "Return the widest live non-side window (the main content area), or nil.
Side windows (the sidebar) are excluded so channels never open into it."
      (let ((wins (seq-remove (lambda (w) (window-parameter w 'window-side))
                              (window-list nil 'no-minibuffer))))
        (car (sort wins (lambda (a b)
                          (> (window-total-width a) (window-total-width b)))))))

    (defun my/clatter--show-channel (buf)
      "Display clatter BUF in the main content area per `my/clatter-channel-open-style'.
Leaves the channel window selected, so focus lands where you type."
      (let ((main (my/clatter--content-window)))
        (cond
         ((and (eq my/clatter-channel-open-style 'split) main)
          (let ((new (with-selected-window main (split-window-sensibly))))
            (select-window (or new main))
            (switch-to-buffer buf)))
         (main
          (select-window main)
          (switch-to-buffer buf))
         ;; No main window yet (only the sidebar): carve one out beside it.
         (t
          (let ((side (get-buffer-window my/clatter-status-buffer-name)))
            (select-window (if side (split-window side nil 'right)
                             (selected-window)))

** Desktop save
Save the desktop periodically and on exit, but never restore it automatically.
=desktop.el= registers an ~after-init-hook~ entry that calls ~desktop-read~ when
~desktop-save-mode~ is on; loading it from ~emacs-startup-hook~ (which runs
later) means that entry either never fires or sees the mode still disabled.

Faking ~desktop-file-modtime~ keeps ~desktop-save~ from asking "Current desktop
was not loaded from a file.  Overwrite?", and the lock is only claimed when no
other Emacs owns it.

Because this session starts saving over the desktop file within seconds, each
startup snapshots the file to =.emacs.desktop.YYYYMMDDThhmmss= (stamped with the
file's own mtime, so restarting without changes doesn't add a duplicate) and
prunes to the newest ~my/desktop-snapshot-keep~.  Restore reads a snapshot, so it
still works after this session has overwritten the live file.

=M-x my/desktop-restore= (also on the dashboard) prompts for a session, annotated
with its buffer count and its buffers grouped by git project — parsed by reading
the snapshot's ~desktop-create-buffer~ forms, never by evaluating them.  A
desktop file stores uniquify's *base* buffer name, so two =README.md= look alike
in it; grouping by project disambiguates them without having to reproduce
whatever ~uniquify-buffer-name-style~ happens to be set to.
Plain ~desktop-read~ refuses with "already loaded" because this Emacs holds the
lock, so the lock is released just before reading; ~desktop-read~ reclaims it.
#+begin_src emacs-lisp :tangle "post-init.el"
  (defvar my/desktop-snapshot-keep 5
    "How many previous sessions to keep desktop snapshots for.")

  (defvar my/desktop-snapshot-regexp "\\.[0-9]\\{8\\}T[0-9]\\{6\\}\\'"
    "Suffix identifying a desktop snapshot of `desktop-base-file-name'.")

  (defvar my/desktop-annotation-max-names 6
    "How many buffer names a snapshot's annotation lists before eliding.")

  (defun my/desktop--snapshot-files ()
    "Desktop snapshots in `desktop-dirname', newest first."
    (and desktop-dirname
         (nreverse
          (directory-files desktop-dirname t
                           (concat "\\`" (regexp-quote desktop-base-file-name)
                                   my/desktop-snapshot-regexp)))))

  (defun my/desktop--snapshot-take ()
    "Snapshot the current desktop file, then prune to `my/desktop-snapshot-keep'."
    (let ((live (desktop-full-file-name)))
      (when (file-exists-p live)
        ;; Stamping with the file's own mtime makes repeated restarts idempotent.
        (let ((dest (concat live
                            (format-time-string
                             ".%Y%m%dT%H%M%S"
                             (file-attribute-modification-time
                              (file-attributes live))))))
          (unless (file-exists-p dest)
            (copy-file live dest t t)))))
    (dolist (old (nthcdr my/desktop-snapshot-keep (my/desktop--snapshot-files)))
      (delete-file old)))

  (defun my/desktop--project-of (file cache)
    "Name of the git project containing FILE, or nil.
  CACHE is a hash table memoizing the lookup per directory."
    (when file
      (let* ((dir (file-name-directory (expand-file-name file)))
             (root (let ((hit (gethash dir cache 'miss)))
                     (if (eq hit 'miss)
                         (puthash dir
                                  (ignore-errors (locate-dominating-file dir ".git"))
                                  cache)
                       hit))))
        (and root (file-name-nondirectory (directory-file-name root))))))

  (defun my/desktop--group-entries (entries)
    "Group ENTRIES, a list of (BUFFER-NAME . FILE), by project.
  Returns a list of (PROJECT . BUFFER-NAMES) in order of first appearance, with
  the buffers belonging to no project last under a nil PROJECT.  Grouping is
  used rather than reconstructing uniquify's names, which would have to assume
  a particular `uniquify-buffer-name-style'."
    (let ((cache (make-hash-table :test #'equal))
          (groups nil))
      (dolist (e entries)
        (let* ((project (my/desktop--project-of (cdr e) cache))
               (cell (assoc project groups)))
          (unless cell
            (setq cell (list project))
            (push cell groups))
          (setcdr cell (cons (car e) (cdr cell)))))
      (setq groups (nreverse groups))
      (dolist (g groups) (setcdr g (nreverse (cdr g))))
      (append (seq-remove (lambda (g) (null (car g))) groups)
              (seq-filter (lambda (g) (null (car g))) groups))))

  (defun my/desktop--summary (file)
    "Summarize desktop snapshot FILE as a plist.
  Keys are :time, :buffers, :projects and :groups.  The buffer forms are read,
  not evaluated, so summarizing a snapshot has no side effects."
    (let ((entries nil))
      (with-temp-buffer
        (insert-file-contents file)
        (goto-char (point-min))
        (while (re-search-forward
                "^(desktop-\\(?:create-buffer\\|append-buffer-args\\) " nil t)
          (ignore-errors
            (read (current-buffer))                  ; file format version
            (let* ((visited (read (current-buffer))) ; visited file, or nil
                   (name (read (current-buffer))))   ; buffer name
              (push (cons (if (stringp name) name "?")
                          (and (stringp visited) visited))
                    entries)))))
      (setq entries (nreverse entries))
      (let ((groups (my/desktop--group-entries entries)))
        (list :time (file-attribute-modification-time (file-attributes file))
              :buffers (length entries)
              :projects (length (seq-remove (lambda (g) (null (car g))) groups))
              :groups groups))))

  (defun my/desktop--relative-age (time)
    "TIME as a short age, e.g. \"3h ago\"."
    (let ((s (float-time (time-subtract nil time))))
      (cond ((< s 60) "just now")
            ((< s 3600) (format "%dm ago" (round (/ s 60))))
            ((< s 86400) (format "%dh ago" (round (/ s 3600))))
            (t (format "%dd ago" (round (/ s 86400)))))))

  (defun my/desktop--label (summary)
    "Completion candidate for SUMMARY."
    (format "%s  (%s)"
            (format-time-string "%Y-%m-%d %H:%M:%S" (plist-get summary :time))
            (my/desktop--relative-age (plist-get summary :time))))

  (defun my/desktop--annotation (summary)
    "Annotation describing the contents of SUMMARY.
  Buffers are listed grouped by project, as
  \"proj1: file1, file2; proj2: file3; file4\", truncated to
  `my/desktop-annotation-max-names' names."
    (let ((budget my/desktop-annotation-max-names)
          (buffers (plist-get summary :buffers))
          (projects (plist-get summary :projects))
          (parts nil)
          (elided nil))
      (dolist (group (plist-get summary :groups))
        (let* ((names (cdr group))
               (shown (seq-take names (max budget 0))))
          (setq budget (- budget (length shown)))
          (unless (= (length shown) (length names)) (setq elided t))
          (when shown
            (push (if (car group)
                      (format "%s: %s" (car group) (string-join shown ", "))
                    (string-join shown ", "))
                  parts))))
      (format "   %d buffer%s · %d project%s · %s%s"
              buffers (if (= buffers 1) "" "s")
              projects (if (= projects 1) "" "s")
              (if parts (string-join (nreverse parts) "; ") "no buffers")
              (if elided " …" ""))))

  (defun my/desktop--read-snapshot ()
    "Prompt for a desktop snapshot and return its file name."
    (let ((files (my/desktop--snapshot-files)))
      (unless files
        (user-error "No desktop snapshots in %s" desktop-dirname))
      (let* ((entries (mapcar (lambda (f)
                                (let ((s (my/desktop--summary f)))
                                  (list (my/desktop--label s) f s)))
                              files))
             (completion-extra-properties
              (list :annotation-function
                    (lambda (key)
                      (let ((e (assoc key entries)))
                        (and e (my/desktop--annotation (nth 2 e)))))))
             (choice (completing-read "Restore desktop session: "
                                      (mapcar #'car entries) nil t nil nil
                                      (caar entries))))
        (nth 1 (assoc choice entries)))))

  (add-hook 'emacs-startup-hook
            (lambda ()
              (require 'desktop)
              (setq desktop-dirname (expand-file-name user-emacs-directory))
              (my/desktop--snapshot-take)
              (desktop--get-file-modtime)
              (let ((owner (desktop-owner)))
                ;; Take the lock unless another live Emacs holds it; auto-saving
                ;; is a no-op unless this process owns the desktop file.
                (unless (and owner (desktop--emacs-pid-running-p owner))
                  (desktop-claim-lock)))
              (desktop-save-mode 1)))

  (defun my/desktop-restore (file)
    "Restore the desktop snapshot FILE, prompting for one interactively.
  `desktop-read' declines to load a desktop whose lock this process already
  holds, which is always the case here, so drop the lock first; the read
  reclaims it."
    (interactive (progn (require 'desktop)
                        (list (my/desktop--read-snapshot))))
    (desktop-release-lock)
    (let ((desktop-base-file-name (file-name-nondirectory file)))
      (desktop-read (file-name-directory file)))
    ;; The read recorded the snapshot's modtime; point the bookkeeping back at
    ;; the live file so the next save does not ask to overwrite it.
    (desktop--get-file-modtime))

  Returns the export time as a number, or nil if not found."
    (when (file-exists-p org-file)
      (with-temp-buffer
        (insert-file-contents org-file)
        (org-mode)
        (goto-char (point-min))
        ;; Search for the property
        (when (re-search-forward "^:KOREADER_EXPORT_TIME:\\s-+\\([0-9]+\\)" nil t)
          (string-to-number (match-string 1))))))

  (defun koreader-get-processed-exports (directory)
    "Scan DIRECTORY for org files and return list of processed export times.
  Returns a list of numbers representing KOREADER_EXPORT_TIME values."
    (let ((org-files (directory-files directory t "\\.org$"))
          (export-times '()))
      (dolist (file org-files)
        (let ((export-time (koreader-get-export-time-from-org file)))
          (when export-time
            (push export-time export-times))))
      export-times))

  (defun koreader-get-json-metadata (json-file)
    "Extract metadata from JSON-FILE without full processing.
  Returns a plist with :title, :author, :created-at, :file, and :json-data.
  The :json-data can be reused to avoid re-reading the file."
    (let* ((json-data (json-read-file json-file))
           (title (alist-get 'title json-data))
           (author (alist-get 'author json-data))
           (created-at (or (alist-get 'created_on json-data) 0)))
      (list :title title
            :author author
            :created-at created-at
            :file json-file
            :json-data json-data)))

  (defun koreader-json-needs-processing-p (metadata processed-exports)
    "Check if file in METADATA needs processing.
  METADATA is a plist with :created-at field.
  PROCESSED-EXPORTS is a list of already-processed export times.
  Returns t if the file needs processing, nil otherwise."
    (let ((created-at (plist-get metadata :created-at)))
      (if (and created-at (> created-at 0))
          (not (member created-at processed-exports))
        ;; If no created_on field, always process
        t)))

  (defun koreader-add-merge-history (org-file json-file new-count skipped-count)
    "Add merge history entry to ORG-FILE's KOREADER_HISTORY drawer.
  Records timestamp, JSON-FILE basename, NEW-COUNT highlights added, and SKIPPED-COUNT existing."
    (when (file-exists-p org-file)
      (with-temp-buffer
        (insert-file-contents org-file)
        (org-mode)
        (goto-char (point-min))

        ;; Find or create the history drawer
        (let ((history-start nil)
              (json-basename (file-name-nondirectory json-file))
              (history-entry (format "- [%s] %s: Imported %d new highlights, %d skipped\n"
                                     (format-time-string "%Y-%m-%d %a %H:%M")
                                     (file-name-nondirectory json-file)
                                     new-count
                                     skipped-count)))

          ;; Search for existing KOREADER_HISTORY drawer
          (if (re-search-forward "^:KOREADER_HISTORY:" nil t)
              ;; Found existing drawer, add entry after the drawer name
              (save-excursion
                (end-of-line)
                (insert "\n" history-entry))

            ;; No drawer found, create one after PROPERTIES drawer or at beginning
            (goto-char (point-min))
            (cond
             ;; Try to insert after PROPERTIES drawer
             ((re-search-forward "^:END:" nil t)
              (end-of-line)
              (insert "\n\n:KOREADER_HISTORY:\n" history-entry ":END:"))
             ;; Otherwise insert after metadata
             (t
              (while (and (not (eobp)) (looking-at "^#\\+"))
                (forward-line 1))
              (insert "\n:KOREADER_HISTORY:\n" history-entry ":END:\n"))))

        ;; Save the modified content
        (write-region (point-min) (point-max) org-file)))))

  (defun koreader-parse-existing-org (org-file)
    "Parse ORG-FILE and return a hash table of existing entries with their content.
  Key format: \"CHAPTER|PAGE|TIMESTAMP\", value is the full entry text."
    (let ((entries-hash (make-hash-table :test 'equal)))
      (when (file-exists-p org-file)
        (with-temp-buffer
          (insert-file-contents org-file)
          (org-mode)
          (goto-char (point-min))
          ;; Skip metadata lines at the beginning
          (while (and (not (eobp)) (looking-at "^#\\+"))
            (forward-line 1))
          (let (current-chapter)
            (while (not (eobp))
              (when (org-at-heading-p)
                (let ((level (org-current-level))
                      (heading (org-get-heading t t t t)))
                  (cond
                   ;; Top-level heading is a chapter
                   ((= level 1)
                    (setq current-chapter heading))
                   ;; Second-level heading is an entry
                   ((and (= level 2) current-chapter)
                    ;; Extract page and timestamp from heading
                    ;; Format: "Page 18 - 2025-09-27 00:05:51"
                    (when (string-match "^Page \\([0-9]+\\) - \\(.+\\)" heading)
                      (let* ((page (string-to-number (match-string 1 heading)))
                             (timestamp (match-string 2 heading))
                             (key (format "%s|%d|%s" current-chapter page timestamp))
                             ;; Get the entire entry content
                             (entry-start (point))
                             (entry-end (save-excursion
                                          (org-end-of-subtree t t)
                                          (point)))
                             (entry-text (buffer-substring-no-properties entry-start entry-end)))
                        (puthash key entry-text entries-hash)))))))
              (forward-line 1)))))
      entries-hash))

  (defun koreader-entry-exists-p (chapter page timestamp entries-hash)
    "Check if an entry exists in ENTRIES-HASH.
  CHAPTER, PAGE, and TIMESTAMP identify the entry.
  Returns the entry text if it exists, nil otherwise."
    (let ((key (format "%s|%d|%s" chapter page timestamp)))
      (gethash key entries-hash)))

  (defun koreader-json-to-org (json-file &optional org-file output-directory json-data denote-timestamp)
    "Convert KOReader JSON-FILE to Org mode format.

  Uses `koreader-use-denote' to determine output format.

  Optional arguments:
  - ORG-FILE: Specific output file path (overrides filename generation)
  - OUTPUT-DIRECTORY: Directory for output file (default: same as JSON-FILE)
  - JSON-DATA: Pre-loaded JSON data to avoid re-reading file
  - DENOTE-TIMESTAMP: Timestamp string for Denote filename (ensures uniqueness)

  Returns a plist with:
  - :file - path to created/updated org file
  - :new-count - number of new highlights added
  - :skipped-count - number of existing highlights skipped
  - :export-time - created_on timestamp from JSON

  Examples:
    ;; Convert a single JSON file to the same directory
    (koreader-json-to-org \"/path/to/book.json\")

    ;; Convert to a specific output directory
    (koreader-json-to-org \"/path/to/book.json\" nil \"~/notes/books/\")

    ;; Specify exact output file
    (koreader-json-to-org \"/path/to/book.json\" \"~/notes/my-book.org\")"
    (interactive "fKOReader JSON file: ")
    (let* ((json-data (or json-data (json-read-file json-file)))
           (title (alist-get 'title json-data))
           (author (alist-get 'author json-data))
           (pages (alist-get 'number_of_pages json-data))
           (created-at (alist-get 'created_on json-data))
           (entries (append (alist-get 'entries json-data) nil)) ;; Convert vector to list
           (target-dir (or output-directory (file-name-directory json-file)))
           (output-file (or org-file
                            (if koreader-use-denote
                                (koreader-generate-denote-filename target-dir title author denote-timestamp)
                              (koreader-generate-plain-filename target-dir title author))))
           (chapters-hash (make-hash-table :test 'equal))
           (existing-entries (koreader-parse-existing-org output-file))
           (new-count 0)
           (skipped-count 0))

      ;; Group entries by chapter
      (dolist (entry entries)
        (let* ((chapter (alist-get 'chapter entry))
               (chapter-entries (gethash chapter chapters-hash)))
          (puthash chapter (append chapter-entries (list entry)) chapters-hash)))

      ;; Write to org file
      (with-temp-file output-file
        ;; Write metadata
        (insert "#+title:       " title "\n")
        (insert "#+date:        " (format-time-string "[%Y-%m-%d %a %H:%M]") "\n")
        (insert "#+book_author: " author "\n")

        ;; Denote-specific metadata
        (when koreader-use-denote
          (insert "#+filetags:    :" koreader-denote-tag ":\n")
          ;; Extract identifier from filename (timestamp part)
          (let ((identifier (if (string-match "^\\([0-9T]+\\)--" (file-name-nondirectory output-file))
                                (match-string 1 (file-name-nondirectory output-file))
                              (format-time-string "%Y%m%dT%H%M%S"))))
            (insert "#+identifier:  " identifier "\n")))

        (insert "\n")

        ;; Properties drawer
        (insert ":PROPERTIES:\n")
        (when created-at
          (insert ":KOREADER_EXPORT_TIME: " (number-to-string created-at) "\n"))
        (insert ":KOREADER_FILE: " json-file "\n")
        (insert ":PAGES: " (number-to-string pages) "\n")
        (insert ":END:\n")

        ;; Write chapters and entries
        (maphash
         (lambda (chapter entries)
           (insert "\n* " chapter "\n")
           (dolist (entry entries)
             (let* ((page (alist-get 'page entry))
                    (timestamp (alist-get 'time entry))
                    (text (alist-get 'text entry))
                    (note (alist-get 'note entry))
                    (human-time (koreader-timestamp-to-human-readable timestamp))
                    (existing-text (koreader-entry-exists-p chapter page human-time existing-entries)))

               ;; Check if entry already exists
               (if existing-text
                   (progn
                     (setq skipped-count (1+ skipped-count))
                     ;; Insert the existing entry as-is
                     (insert existing-text))
                 ;; Entry is new, add it
                 (setq new-count (1+ new-count))

                 ;; Entry heading with page number, timestamp, and tags
                 (insert "** Page " (number-to-string page) " - " human-time)
                 (when koreader-add-entry-tags
                   (let ((tag (if note
                                  (concat ":" koreader-note-tag ":")
                                (concat ":" koreader-highlight-tag ":"))))
                     (insert " " tag)))
                 (insert "\n")

                 ;; Quoted text if present
                 (when text
                   (insert "#+BEGIN_QUOTE\n")
                   (insert text "\n")
                   (insert "#+END_QUOTE\n"))

                 ;; Personal note if present
                 (when note
                   (insert "\n")
                   (insert note "\n"))

                 (insert "\n")))))
         chapters-hash))

      ;; Add merge history
      (koreader-add-merge-history output-file json-file new-count skipped-count)

      (message "Converted %s to %s (%d new, %d skipped)"
               json-file output-file new-count skipped-count)
      (when (called-interactively-p 'any)
        (find-file output-file))

      ;; Return detailed results
      (list :file output-file
            :new-count new-count
            :skipped-count skipped-count
            :export-time created-at)))

  (defun koreader-process-json-directory (json-directory output-directory)
    "Process all JSON files in JSON-DIRECTORY and output to OUTPUT-DIRECTORY.

  For each book (title+author combination), only processes the LATEST export file,
  as KOReader exports are cumulative. Skips books whose latest export has already
  been processed (based on created_on timestamp).

  Uses `koreader-use-denote' to determine the output format.

  Returns a plist with summary statistics:
  - :files-processed - number of JSON files processed
  - :files-skipped - number of JSON files skipped (old exports or already processed)
  - :total-new - total new highlights across all files
  - :total-skipped - total existing highlights across all files
  - :results - list of individual file results

  Examples:
    ;; Process all JSON files from KOReader export directory
    (koreader-process-json-directory
     \"~/koreader/exports/\"
     \"~/notes/books/\")

    ;; Check results
    (let ((results (koreader-process-json-directory
                    \"~/koreader/exports/\"
                    \"~/notes/books/\")))
      (message \"Processed %d books with %d new highlights\"
               (plist-get results :files-processed)
               (plist-get results :total-new)))"
    (let* ((json-files (directory-files json-directory t "\\.json$"))
           (processed-exports (koreader-get-processed-exports output-directory))
           (books-hash (make-hash-table :test 'equal))
           (files-to-process '())
           (files-skipped 0)
           (total-new 0)
           (total-skipped 0)
           (results '()))

      ;; Group files by book and find latest export for each
      (dolist (json-file json-files)
        (let* ((metadata (koreader-get-json-metadata json-file))
               (title (plist-get metadata :title))
               (author (plist-get metadata :author))
               (created-at (plist-get metadata :created-at))
               (book-key (concat title "|" author))
               (existing (gethash book-key books-hash)))

          ;; Debug output
          (when koreader-debug-logging
            (message "File: %s | created_at: %s"
                     (file-name-nondirectory json-file)
                     created-at))

          ;; Keep only the latest export for each book
          (when (or (not existing)
                    (> created-at (plist-get existing :created-at)))
            (when koreader-debug-logging
              (message "  -> Selected as latest for: %s" book-key))
            (puthash book-key metadata books-hash))))

      ;; Filter books that need processing (latest export not yet processed)
      (maphash
       (lambda (book-key metadata)
         (if (koreader-json-needs-processing-p metadata processed-exports)
             (push metadata files-to-process)
           (setq files-skipped (1+ files-skipped))))
       books-hash)

      ;; Count files that were skipped because they're old exports
      (setq files-skipped (+ files-skipped
                             (- (length json-files)
                                (hash-table-count books-hash))))

      ;; Process each file using cached JSON data
      ;; Track timestamp to ensure uniqueness for Denote filenames
      (let ((current-time (current-time)))
        (dolist (metadata files-to-process)
          (let* ((json-file (plist-get metadata :file))
                 (json-data (plist-get metadata :json-data))
                 (timestamp (format-time-string "%Y%m%dT%H%M%S" current-time)))
            (condition-case err
                (let ((result (koreader-json-to-org json-file nil output-directory json-data timestamp)))
                  (setq total-new (+ total-new (plist-get result :new-count)))
                  (setq total-skipped (+ total-skipped (plist-get result :skipped-count)))
                  (push result results))
              (error
               (message "Error processing %s: %s" json-file (error-message-string err))))
            ;; Increment time by 1 second for next file
            (setq current-time (time-add current-time (seconds-to-time 1))))))

      ;; Return summary
      (list :files-processed (length files-to-process)
            :files-skipped files-skipped
            :total-new total-new
            :total-skipped total-skipped
            :results (nreverse results))))
